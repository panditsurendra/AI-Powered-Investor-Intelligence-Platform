        // Clean & parse financial numeric values for dynamic bars sizing
        function parseNumericValue(valStr) {
            if (!valStr) return 0;
            let cleaned = valStr.replace(/[$,\sM]/g, '');
            let isNegative = false;
            if (cleaned.startsWith('(') && cleaned.endsWith(')')) {
                isNegative = true;
                cleaned = cleaned.slice(1, -1);
            }
            let num = parseFloat(cleaned);
            return isNaN(num) ? 0 : (isNegative ? -num : num);
        }

        // Scaler algorithm to determine dynamic bar widths relative to card maximums
        function recalculateBarScaling() {
            const cards = document.querySelectorAll('.kpi-card');
            
            cards.forEach(card => {
                const rows = card.querySelectorAll('.kpi-value-row');
                let maxVal = 0;
                const parsedVals = [];

                rows.forEach(row => {
                    const rawValText = row.querySelector('.kpi-val-number').textContent.trim();
                    const num = parseNumericValue(rawValText);
                    parsedVals.push(num);
                    if (Math.abs(num) > maxVal) {
                        maxVal = Math.abs(num);
                    }
                });

                rows.forEach((row, idx) => {
                    const num = parsedVals[idx];
                    const barFill = row.querySelector('.kpi-compare-bar-fill');
                    if (barFill) {
                        let pct = maxVal > 0 ? (Math.abs(num) / maxVal) * 100 : 10;
                        barFill.style.width = `${pct}%`;

                        // Add distinct dynamic indicators for negative yields
                        if (num < 0) {
                            barFill.style.background = 'linear-gradient(to right, var(--accent-red), rgba(244,63,94,0.4))';
                        } else {
                            const metricType = card.getAttribute('data-metric');
                            let color = 'var(--accent-blue)';
                            if (metricType === 'revenue') color = 'var(--accent-blue)';
                            else if (metricType === 'net_income') color = 'var(--accent-green)';
                            else if (metricType === 'operating_income') color = 'var(--accent-amber)';
                            else if (metricType === 'cash_flow') color = 'var(--accent-purple)';
                            else if (metricType === 'total_assets') color = '#06b6d4';
                            else if (metricType === 'total_liabilities') color = 'var(--accent-red)';
                            
                            barFill.style.background = `linear-gradient(to right, ${color}, rgba(255,255,255,0.03))`;
                        }
                    }
                });
            });
        }

        // Audio helper to create futuristic sounds without external URLs
        function playBeep(frequency, type, duration) {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                
                osc.type = type;
                osc.frequency.value = frequency;
                gain.gain.setValueAtTime(0.06, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.00001, ctx.currentTime + duration);
                
                osc.connect(gain);
                gain.connect(ctx.destination);
                
                osc.start();
                osc.stop(ctx.currentTime + duration);
            } catch(e) {
                console.warn("Audio Context not supported or allowed yet");
            }
        }

        // Chat Conversation Handler
        function sendMessage() {
            const input = document.getElementById('chatInput');
            const text = input.value.trim();
            if (!text) return;

            // Render User bubble
            renderChatBubble(text, 'user');
            input.value = "";

            // Auto Scroll

            // Render Bot typing state

            scrollChatToBottom();

            const selectedCompany = document.getElementById('chatCompanySelect').value;

            // Mock conversational responses tailored around loaded dataset
            setTimeout(() => {
                // Destroy typing bubble
                const typingBubble = document.getElementById(typingId);
                if (typingBubble) typingBubble.remove();

                let responseHTML = "";
                const normalizedText = text.toLowerCase();

                if (selectedCompany) {
                    const data = metricsData.find(d => d.company === selectedCompany);
                    if (normalizedText.includes('revenue') || normalizedText.includes('sales')) {
                        responseHTML = `<p>Based on the analyzed reports, <strong>${data.company}</strong> reported a total revenue of <strong>${data.revenue}</strong> for the fiscal year <strong>${data.year}</strong>.</p>`;
                    } else if (normalizedText.includes('risk') || normalizedText.includes('liability')) {
                        responseHTML = `<p>My analysis of <strong>${data.company} (${data.year})</strong> shows these top liabilities or structural risks:</p>
                        <ul style="margin: 8px 0 0 18px; padding: 0;">
                            ${data.risk_factors.map(r => `<li style="margin-bottom:4px;">${r}</li>`).join('')}
                        </ul>`;
                    } else if (normalizedText.includes('growth') || normalizedText.includes('driver')) {
                        responseHTML = `<p>Here are the qualitative growth drivers extracted for <strong>${data.company}</strong>:</p>
                        <ul style="margin: 8px 0 0 18px; padding: 0;">
                            ${data.growth_drivers.map(g => `<li style="margin-bottom:4px;">${g}</li>`).join('')}
                        </ul>`;
                    } else {
                        responseHTML = `<p>Regarding <strong>${data.company} (${data.year})</strong>: its operating income stands at ${data.operating_income} alongside operating cash flow of ${data.cash_flow}. Let me know if you need specific details on its growth strategies or risk profile.</p>`;
                    }
                } else {
                    // Global queries
                    if (normalizedText.includes('compare') || normalizedText.includes('highest')) {
                        const highestCompany = metricsData.reduce((prev, current) => {
                            return parseNumericValue(prev.revenue) > parseNumericValue(current.revenue) ? prev : current;
                        });
                        responseHTML = `<p>Comparing our current dataset: <strong>${highestCompany.company}</strong> holds the highest revenue at <strong>${highestCompany.revenue}</strong>, followed by ${metricsData.filter(d => d.company !== highestCompany.company).map(d => `${d.company} (${d.revenue})`).join(', ')}.</p>`;
                    } else {
                        responseHTML = `<p>I can help you review and compare metrics across our ${metricsData.length} indexed corporate profiles. Please select a specific company context on the selector above or ask me about global revenue rankings!</p>`;
                    }
                }

                renderChatBubble(responseHTML, 'bot');
                scrollChatToBottom();
            }, 1200);
        }
